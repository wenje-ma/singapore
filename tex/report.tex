\begin{frame}{The Problem: Few Expensive Tests}
  Your job: tune parameters, but you face:
  \begin{itemize}
    \item Only about \textbf{fifteen expensive} experiments allowed
    \item \textbf{Many interacting} parameters --- more than ten
    \item Change one, the others move --- can't tune one at a time
    \item So: spend a tiny budget in the \textbf{smartest way}?
  \end{itemize}
\end{frame}

\begin{frame}{Why It Is Hard: Black Box \& Budget}
  \begin{itemize}
    \item \textbf{Black box}: you see input and output, never the inside
    \item \textbf{Budget}: only fifteen precise runs, but a huge space
    \item Task: find the \textbf{best point} of a high-dimensional black box
  \end{itemize}
\end{frame}

\begin{frame}{The Key Idea: Two Channels}
  Good news --- our black box has two channels:
  \begin{equation*}
    h_k(\boldsymbol{x}) = h_{k-1}(\boldsymbol{x}) + \delta_k(\boldsymbol{x})
  \end{equation*}
  \begin{itemize}
    \item \textbf{Cheap} channel: run many times, but \textbf{biased}
    \item \textbf{Expensive} channel: only 15 runs, accurate
    \item Idea: combine --- accurate = cheap + a correction
  \end{itemize}
\end{frame}

\begin{frame}{Tool 1: Maxpro Design}
  \begin{columns}
    \column{0.55\textwidth}
      Maxpro criterion:
      \begin{equation*}
        \min_D \sum_{i=1}^{n}\prod_{j\neq i}\frac{1}{\sum_{k=1}^p (x_{ik}-x_{jk})^2}
      \end{equation*}
      \begin{itemize}
        \item Start \textbf{evenly}: spread out in \textbf{every projection}
      \end{itemize}
    \column{0.45\textwidth}
      \begin{figure}
        \includegraphics[width=\textwidth]{../figures/2}
      \end{figure}
  \end{columns}
\end{frame}

\begin{frame}{Tool 2: The Stand-In Model (Gaussian Process)}
  \begin{itemize}
    \item A \textbf{Gaussian process} is a normal distribution over \textbf{whole functions} --- not one number
    \item It predicts a value \textbf{and} an \textbf{uncertainty} at every point
    \item Uncertainty tells us \textbf{where to test next}: hopeful \& unsure
  \end{itemize}
\end{frame}

\begin{frame}{The Whole Pipeline}
  \begin{enumerate}
    \item \textbf{Spread out} the starting points
    \item Feed in many \textbf{cheap} runs $\rightarrow$ a \textbf{rough map}
    \item Send \textbf{one expensive} run to the predicted valley
    \item \textbf{Repeat} --- the map gets sharper every round
  \end{enumerate}
\end{frame}

\begin{frame}{Our Question}
  \begin{equation*}
    \widehat{\boldsymbol{x}} \approx \argmin_{\boldsymbol{x}\in[0,1]^p} h_\mathrm{H}(\boldsymbol{x})
  \end{equation*}
  \begin{itemize}
    \item Goal: minimize the expensive function with only \textbf{15 precise runs}
    \item Three small questions:
      \begin{itemize}
        \item Does \textbf{combining the channels} help?
        \item Does \textbf{screening} the parameters help?
        \item Does it still work \textbf{in high dimensions}?
      \end{itemize}
  \end{itemize}
\end{frame}

  \begin{columns}

      \end{figure}

        \item Metric: \textbf{median} of the best found over 20 repeats --- lower is better
        \item Cheap + precise \textbf{beats} precise alone
        \item 1-D: precise alone stuck at 0; with fusion $\rightarrow$ \textbf{global optimum}

      \end{itemize}
  \end{columns}

\begin{frame}{Answer 2: Screening Is Useless}
  \begin{columns}
    \column{0.5\textwidth}
      \begin{figure}
        \includegraphics[width=\textwidth]{../figures/4}
      \end{figure}
    \column{0.5\textwidth}
      \begin{itemize}
        \item Screening \textbf{ranks} parameters by cheap runs
        \item But cheap runs are \textbf{biased} $\rightarrow$ ranking is unreliable
        \item Result: \textbf{removing} screening actually helps
      \end{itemize}
  \end{columns}
\end{frame}
